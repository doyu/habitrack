# badge = checked days in the trailing 30 days, as n/30; a miss does not reset it
with tempfile.TemporaryDirectory() as d:
    p = Path(d) / "ledger.csv"
    t = datetime.now(TZ).date()
    days = [t - timedelta(days=i) for i in (1, 3, 31)]  # 31 days ago is outside the window
    p.write_text("date,x\n" + "".join(f"{day.isoformat()},True\n" for day in sorted(days)))
    cli = Client(create_app(p, "pw", "k"))
    cli.post("/login", data={"password": "pw"})
    assert '<span class="count">2/30</span>' in cli.get("/").text
# click-to-edit rename; a leading ~ archives, removing it restores; invalid names re-render the form with the error
with tempfile.TemporaryDirectory() as d:
    p = Path(d) / "ledger.csv"
    p.write_text("date,読書\n2026-08-23,True\n")
    cli = Client(create_app(p, "pw", "k"))
    cli.post("/login", data={"password": "pw"})
    header = lambda: p.read_text().splitlines()[0]

    assert 'hx-get="/edit/%E8%AA%AD%E6%9B%B8"' in cli.get("/").text  # the name is clickable
    r = cli.get("/edit/%E8%AA%AD%E6%9B%B8").text
    assert 'value="読書"' in r and 'hx-post="/rename/%E8%AA%AD%E6%9B%B8"' in r

    assert 'id="ledger"' in cli.post("/rename/%E8%AA%AD%E6%9B%B8", data={"new": "本"}).text
    assert header() == "date,本"
    cli.post("/rename/%E6%9C%AC", data={"new": "~本"})  # archive by typing ~
    assert header() == "date,~本"
    r = cli.get("/edit/%7E%E6%9C%AC").text
    assert 'value="~本"' in r  # archived row shows its marker
    cli.post("/rename/%7E%E6%9C%AC", data={"new": "~読書"})  # rename while archived stays archived
    assert header() == "date,~読書"
    cli.post("/rename/%7E%E8%AA%AD%E6%9B%B8", data={"new": "読書"})  # restore by removing ~
    assert header() == "date,読書"

    r = cli.post("/rename/%E8%AA%AD%E6%9B%B8", data={"new": "a/b"}).text
    assert "forbidden" in r and 'value="a/b"' in r and header() == "date,読書"  # error shown, file untouched
    assert p.read_text().count("\n") == 2  # history row still there
